"""直接运行启动GUI，或使用命令行执行可复现任务。"""
import argparse
from pathlib import Path
import sys
from uuid import uuid4
import webbrowser

def main():
    parser=argparse.ArgumentParser(description='头皮脑电癫痫发作预测软件 本地研究软件')
    parser.add_argument('--port',type=int,default=8765)
    parser.add_argument('--no-browser',action='store_true')
    parser.add_argument('--action',choices=['demo','prepare','train','predict','evaluate','history','connectivity'])
    parser.add_argument('--dataset'); parser.add_argument('--model'); parser.add_argument('--predictions')
    parser.add_argument('--manifest'); parser.add_argument('--project',default='research_results')
    parser.add_argument('--threshold',type=float,default=.5)
    parser.add_argument('--source',choices=['demo','edf'],default='demo')
    parser.add_argument('--edf-path')
    parser.add_argument('--start-seconds',type=float,default=30)
    parser.add_argument('--duration-seconds',type=float,default=120)
    args=parser.parse_args()
    root=Path(__file__).resolve().parent
    try:
        if args.action:
            from eegseizguard.pipeline import run_action
            import threading
            import json
            out=root/'outputs'/('cli-'+uuid4().hex[:12])
            def log(message,progress=None):
                print(message,flush=True)
            params={k:v for k,v in vars(args).items() if v is not None}
            if args.action=='connectivity':
                params={k:params[k] for k in ('source','edf_path','start_seconds','duration_seconds') if k in params}
            result=run_action(args.action,params,out,log,threading.Event())
            print(json.dumps(dict(output=str(out),summary=result['summary']),ensure_ascii=False),flush=True)
            return 0
        from eegseizguard.server import LocalServer
        if not 0<=args.port<=65535:
            raise ValueError('端口需在0到65535之间。')
        try:
            server=LocalServer(root,args.port)
        except OSError:
            if args.port!=8765:
                raise
            server=LocalServer(root,0)
        url=f'http://127.0.0.1:{server.server_port}'
        print(f'头皮脑电癫痫发作预测软件 V1.0\n{url}\nClose this window or press Ctrl+C to stop.',flush=True)
        if not args.no_browser:
            webbrowser.open(url)
        try:
            server.serve_forever(poll_interval=.3)
        except KeyboardInterrupt:
            pass
        finally:
            for event in list(server.manager.cancel_events.values()):
                event.set()
            server.server_close()
        return 0
    except ModuleNotFoundError as exc:
        print(f'缺少运行依赖：{exc.name}。请执行 python -m pip install -r requirements.txt。',file=sys.stderr)
    except Exception as exc:
        print(f'启动/任务失败：{type(exc).__name__}: {exc}',file=sys.stderr)
    return 1

if __name__=='__main__':
    sys.exit(main())

"""matplotlib 图表生成：ROC/PR/患者对比/混淆矩阵，产物为 PNG 并纳入任务产物。

图表仅用于界面展示与结果归档，指标一律以 metrics.json 数值为准。

import io

import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
import numpy as np

plt.rcParams['font.sans-serif'] = ['Microsoft YaHei', 'SimHei', 'SimSun']
plt.rcParams['axes.unicode_minus'] = False

def _save(fig, path):
    buf = io.BytesIO()
    fig.savefig(buf, format='png', dpi=110, bbox_inches='tight')
    plt.close(fig)
    atomic_bytes(path, buf.getvalue())


def _pooled(metrics, key):
    value = (metrics.get('pooled') or {}).get(key)
    return None if value is None else float(value)


def roc_chart(metrics, path):
    points = (metrics.get('curves') or {}).get('roc')
    if not points:
        return None
    fig, ax = plt.subplots(figsize=(5.4, 4.4))
    x = np.asarray([p[0] for p in points]); y = np.asarray([p[1] for p in points])
    auc = _pooled(metrics, 'auroc')
    label = 'ROC' if auc is None else f'ROC (AUC = {auc:.4f})'
    ax.plot(x, y, color='#169d7d', lw=1.8, label=label)
    ax.plot([0, 1], [0, 1], ls='--', color='#b7c4c0', lw=1, label='随机基线')
    ax.set_xlim(0, 1); ax.set_ylim(0, 1.02)
    ax.set_xlabel('假阳性率 (FPR)'); ax.set_ylabel('真阳性率 (TPR)')
    ax.set_title('ROC 曲线（合并已知标签窗口）'); ax.legend(loc='lower right', fontsize=9)
    ax.grid(alpha=.25, lw=.5)
    _save(fig, path)
    return path.name


def pr_chart(metrics, path):
    points = (metrics.get('curves') or {}).get('pr')
    if not points:
        return None
    fig, ax = plt.subplots(figsize=(5.4, 4.4))
    recall = np.asarray([p[0] for p in points]); precision = np.asarray([p[1] for p in points])
    ap = _pooled(metrics, 'ap')
    baseline = float((metrics.get('pooled') or {}).get('prevalence') or 0)
    label = 'PR' if ap is None else f'PR (AP = {ap:.4f})'
    ax.plot(recall, precision, color='#2b6cb0', lw=1.8, label=label)
    ax.axhline(baseline, ls='--', color='#b7c4c0', lw=1, label=f'阳性率基线 ({baseline:.4f})')
    ax.set_xlim(0, 1); ax.set_ylim(0, 1.02)
    ax.set_xlabel('召回率 (Recall)'); ax.set_ylabel('精确率 (Precision)')
    ax.set_title('PR 曲线（合并已知标签窗口）'); ax.legend(loc='upper right', fontsize=9)
    ax.grid(alpha=.25, lw=.5)
    _save(fig, path)
    return path.name


def patient_chart(metrics, path):
    rows = metrics.get('patients')
    if not rows:
        return None
    usable = [r for r in rows if r.get('auroc') is not None or r.get('ap') is not None]
    if not usable:
        return None
    names = [str(r['patient']) for r in usable]
    auc = [float(r['auroc']) if r.get('auroc') is not None else np.nan for r in usable]
    ap = [float(r['ap']) if r.get('ap') is not None else np.nan for r in usable]
    fig, axes = plt.subplots(1, 2, figsize=(8.4, 4.0))
    for ax, values, title, color in ((axes[0], auc, '各患者 AUROC', '#169d7d'),
                                     (axes[1], ap, '各患者 AP', '#2b6cb0')):
        ax.barh(names, values, color=color, alpha=.85)
        ax.set_xlim(0, 1)
        ax.set_xlabel(title)
        ax.grid(axis='x', alpha=.25, lw=.5)
        for i, v in enumerate(values):
            if not np.isnan(v):
                ax.text(min(v + .01, .98), i, f'{v:.4f}', ha='right' if v>.85 else 'left', va='center', fontsize=8)
    fig.suptitle('患者指标对比（宏平均为各患者等权）', y=1.02, fontsize=11)
    fig.tight_layout()
    _save(fig, path)
    return path.name


def confusion_chart(metrics, path):
    matrix = (metrics.get('pooled') or {}).get('confusion')
    if not matrix or len(matrix) != 2:
        return None
    m = np.asarray(matrix, dtype=int)
    fig, ax = plt.subplots(figsize=(4.4, 4.0))
    im = ax.imshow(m, cmap='Blues')
    labels = [['真阴性 TN', '假阳性 FP'], ['假阴性 FN', '真阳性 TP']]
    for i in range(2):
        for j in range(2):
            ax.text(j, i, f'{labels[i][j]}\n{m[i, j]}', ha='center', va='center',
                    fontsize=12, color='white' if m[i, j] > m.max() / 2 else '#264653')
    ax.set_xticks([0, 1]); ax.set_yticks([0, 1])
    ax.set_xticklabels(['预测阴性', '预测阳性'])
    ax.set_yticklabels(['真实阴性', '真实阳性'])
    ax.set_title('混淆矩阵（合并窗口）')
    fig.colorbar(im, ax=ax, fraction=.046)
    _save(fig, path)
    return path.name


def make_charts(metrics, out_dir, warnings=None):
    """为评估指标生成全部图表，返回生成的文件名列表；缺数据时跳过对应图。"""
    if not metrics:
        return []
    names = []
    for factory, filename in ((roc_chart, 'roc_curve.png'), (pr_chart, 'pr_curve.png'),
                              (patient_chart, 'patient_metrics.png'),
                              (confusion_chart, 'confusion_matrix.png')):
        try:
            name = factory(metrics, out_dir / filename)
            if name:
                names.append(name)
        except Exception as exc:
            plt.close('all')
            if warnings is not None:
                warnings.append(f'{filename} 绘制失败：{type(exc).__name__}: {exc}')
            continue
    return names

"""固定、可追溯的信号与标签协议。"""
from dataclasses import asdict, dataclass
import hashlib
import json

CHANNELS = tuple(('FP1-F7 F7-T7 T7-P7 P7-O1 FP1-F3 F3-C3 C3-P3 P3-O1 '
                  'FP2-F4 F4-C4 C4-P4 P4-O2 FP2-F8 F8-T8 T8-P8 P8-O2 FZ-CZ CZ-PZ').split())
BANDS = ((0.5, 4), (4, 8), (8, 13), (13, 30), (30, 45))
FEATURE_SCHEMA = 'welch_log10_uv2_hann_2s_boundary_interp_channel_major_v1'

@dataclass(frozen=True)
class Protocol:
    sampling_hz: int = 128
    window_seconds: int = 10
    stride_seconds: int = 10
    warmup_seconds: int = 30
    sph_seconds: int = 300
    sop_seconds: int = 1800
    postictal_seconds: int = 1800
    protocol_id: str = 'recording_conditional_r1'

    def to_dict(self):
        return asdict(self)

    @property
    def horizon(self):
        return self.sph_seconds + self.sop_seconds

PROTOCOL = Protocol()

def feature_contract():
    return dict(feature_schema=FEATURE_SCHEMA, channels=list(CHANNELS),
                bands_hz=[list(b) for b in BANDS], protocol=PROTOCOL.to_dict())

def canonical_hash(value):
    return hashlib.sha256(json.dumps(value, sort_keys=True, ensure_ascii=False,
                                    separators=(',', ':'), allow_nan=False).encode('utf-8')).hexdigest()

"""从真实时序估计头皮双极导联相干性；不推断皮层网络或发作风险。

采用固定 Welch 参数与有界十秒窗口。三维坐标只是标准电极示意位置的
中点方向，不能替代个体影像、源定位或电极实测位置。
"""
import csv
import hashlib
import io
import math
from pathlib import Path

import numpy as np
from scipy import signal

from .config import BANDS, CHANNELS, PROTOCOL
from .signal import CausalFilter, EDFReader
from .storage import atomic_bytes, check_cancel, input_file, write_json

FS = PROTOCOL.sampling_hz
WIDTH = FS * PROTOCOL.window_seconds
BAND_IDS = ('delta', 'theta', 'alpha', 'beta', 'gamma')
BAND_LABELS = ('δ Delta', 'θ Theta', 'α Alpha', 'β Beta', 'γ Gamma')
MAX_FRAMES = 60
DEMO_SEED = 1701


def network_nodes():
    """球面位置 x 向右、y 向上、z 向前，仅供空间布局。"""
    electrodes = {
        'FP1': (-.30, .43, .85), 'FP2': (.30, .43, .85),
        'F7': (-.82, .20, .52), 'F8': (.82, .20, .52),
        'F3': (-.44, .72, .45), 'F4': (.44, .72, .45),
        'C3': (-.58, .80, 0), 'C4': (.58, .80, 0),
        'T7': (-.98, .12, 0), 'T8': (.98, .12, 0),
        'P7': (-.82, .20, -.52), 'P8': (.82, .20, -.52),
        'P3': (-.44, .72, -.45), 'P4': (.44, .72, -.45),
        'O1': (-.30, .43, -.85), 'O2': (.30, .43, -.85),
        'FZ': (0, .85, .53), 'CZ': (0, 1, 0), 'PZ': (0, .85, -.53),
    }
    nodes = []
    for index, name in enumerate(CHANNELS):
        pair = name.split('-')
        positions = [np.asarray(electrodes[e], dtype=float) for e in pair]
        midpoint = sum(p / np.linalg.norm(p) for p in positions)
        midpoint /= np.linalg.norm(midpoint)
        hemisphere = 'midline' if abs(midpoint[0]) < .001 else ('left' if midpoint[0] < 0 else 'right')
        region = ('额侧' if midpoint[2] > .35 else ('枕顶侧' if midpoint[2] < -.35 else '中央侧'))
        nodes.append(dict(id=name, index=index, label=name, electrodes=pair,
                          position=np.round(midpoint, 6).tolist(),
                          hemisphere=hemisphere, region=region))
    return nodes


def _json_numbers(values, decimals=7):
    values = np.round(np.asarray(values, dtype=float), decimals)
    result = values.astype(object)
    result[~np.isfinite(values)] = None
    return result.tolist()


def analyze_window(samples_uv, start_seconds):
    """返回一帧。不可估计的相干性用 null；频带为左闭右开，末带含45Hz。"""
    x = np.asarray(samples_uv, dtype=np.float64)
    if x.shape != (len(CHANNELS), WIDTH) or not np.isfinite(x).all():
        raise ValueError('网络分析需要18导联、128Hz、完整10秒的有限时序。')
    if np.max(np.abs(x)) > 1e12:
        raise ValueError('信号量级超过数值处理预算，请核验EDF物理单位与校准。')
    if isinstance(start_seconds, bool) or not isinstance(start_seconds, (int, float)) or not math.isfinite(start_seconds):
        raise ValueError('网络窗口起点必须为有限秒数。')
    # 与 scipy.signal.coherence 使用相同的分段、去均值和周期 Hann 窗。
    segments = np.lib.stride_tricks.sliding_window_view(x, 256, axis=1)[:, ::128, :]
    centered = segments - segments.mean(axis=-1, keepdims=True)
    taper = signal.get_window('hann', 256, fftbins=True)
    transformed = np.fft.rfft(centered * taper, axis=-1)
    cross = np.einsum('isf,jsf->ijf', transformed.conj(), transformed, optimize=True) / segments.shape[1]
    power = np.real(np.diagonal(cross, axis1=0, axis2=1)).T.copy()
    denominator = power[:, None, :] * power[None, :, :]
    coherence = np.full_like(denominator, np.nan)
    valid_channels = np.std(x, axis=1) > 1e-8
    # 只剔除相对于本导联最大谱功率低于浮点量级的频点，不人为提高连接。
    meaningful = power > np.maximum(np.max(power, axis=1, keepdims=True) * 1e-12, np.finfo(float).tiny)
    valid_bins = meaningful[:, None, :] & meaningful[None, :, :]
    valid_bins &= valid_channels[:, None, None] & valid_channels[None, :, None]
    np.divide(np.abs(cross) ** 2, denominator, out=coherence, where=valid_bins)
    np.clip(coherence, 0, 1, out=coherence)
    frequencies = np.fft.rfftfreq(256, d=1 / FS)
    psd = power / (FS * np.sum(taper ** 2))
    psd[:, 1:-1] *= 2
    matrices, bandpower = {}, {}
    for index, (name, (lo, hi)) in enumerate(zip(BAND_IDS, BANDS)):
        mask = (frequencies >= lo) & ((frequencies <= hi) if index == 4 else (frequencies < hi))
        selected = coherence[:, :, mask]
        counts = np.isfinite(selected).sum(axis=-1)
        average = np.divide(np.nansum(selected, axis=-1), counts,
                            out=np.full((18, 18), np.nan), where=counts > 0)
        matrices[name] = _json_numbers(average)
        # 功率与原模型保持同一频带积分边界，但此处展示线性 µV²。
        grid = np.r_[lo, frequencies[(frequencies > lo) & (frequencies < hi)], hi]
        values = np.stack([np.interp(grid, frequencies, row) for row in psd])
        bandpower[name] = _json_numbers(np.trapezoid(values, grid, axis=-1))
    # 保留全部128Hz采样；简单抽点会让高频EEG在波形视图中混叠。
    display_indices = np.arange(WIDTH)
    psd_visible = frequencies <= 45
    flat = [CHANNELS[i] for i in np.flatnonzero(~valid_channels)]
    warnings = ['平坦导联不参与连接估计：' + '、'.join(flat)] if flat else []
    return dict(start_seconds=float(start_seconds), end_seconds=float(start_seconds + 10),
                matrices=matrices, bandpower=bandpower,
                waveform=dict(t=_json_numbers(display_indices / FS),
                              samples=_json_numbers(x[:, display_indices], 4),
                              units='µV', display_sampling_hz=FS,
                              note='完整128Hz采样，与相干性输入一致；屏幕像素不足时应缩短显示时间范围。'),
                psd=dict(frequency_hz=_json_numbers(frequencies[psd_visible]),
                         values=_json_numbers(psd[:, psd_visible]), units='µV²/Hz'),
                quality=dict(valid_channels=valid_channels.tolist(), flat_channels=flat,
                             warnings=warnings, flat_std_threshold_uv=1e-8))


def _validated_params(params):
    if not isinstance(params, dict):
        raise ValueError('网络分析参数必须为对象。')
    unknown = set(params) - {'source', 'edf_path', 'start_seconds', 'duration_seconds'}
    if unknown:
        raise ValueError('无法识别的网络参数：' + '、'.join(sorted(unknown)))
    source = params.get('source', 'demo')
    if not isinstance(source, str) or source not in {'demo', 'edf'}:
        raise ValueError('网络数据来源必须为 demo 或 edf。')
    values = []
    for key, default in [('start_seconds', 30), ('duration_seconds', 120)]:
        value = params.get(key, default)
        if isinstance(value, bool) or not isinstance(value, (int, float)) or not math.isfinite(value) or value % 10:
            raise ValueError(f'{key} 必须为10的整数倍秒数。')
        values.append(int(value))
    start, duration = values
    if not 30 <= start <= 86400 - 10:
        raise ValueError('起点需在30秒预热之后，且在24小时记录预算内。')
    if not 10 <= duration <= MAX_FRAMES * 10:
        raise ValueError('分析时长须为10至600秒，最多60个窗口。')
    if start + duration > 86400:
        raise ValueError('分析终点超过24小时记录预算。')
    if source == 'demo' and start + duration > 3600:
        raise ValueError('合成功能演示的终点最多为3600秒。')
    if source == 'demo' and params.get('edf_path'):
        raise ValueError('演示模式不能同时指定EDF文件，以免混淆数据来源。')
    return source, start, duration


def _sha256(path, cancel):
    digest = hashlib.sha256()
    with path.open('rb') as stream:
        for chunk in iter(lambda: stream.read(1024 * 1024), b''):
            check_cancel(cancel)
            digest.update(chunk)
    return digest.hexdigest()


def _frames_from_blocks(blocks, native_fs, start, duration, log, cancel):
    """从记录零点滤波，仅保留分析段和不足一帧的缓冲。"""
    causal = CausalFilter(native_fs)
    beginning, finish = start * FS, (start + duration) * FS
    seen, buffer, frames = 0, np.empty((18, 0), dtype=np.float32), []
    for block in blocks:
        check_cancel(cancel)
        volts = (block * 1e-6).astype(np.float32).astype(np.float64)
        filtered = (causal.transform(volts) * 1e6).astype(np.float32)
        a, b = max(beginning - seen, 0), min(finish - seen, filtered.shape[1])
        if b > a:
            buffer = np.concatenate((buffer, filtered[:, a:b]), axis=1)
        seen += filtered.shape[1]
        while buffer.shape[1] >= WIDTH:
            check_cancel(cancel)
            frames.append(analyze_window(buffer[:, :WIDTH], start + 10 * len(frames)))
            buffer = buffer[:, WIDTH:]
            log(f'计算第 {len(frames)}/{duration // 10} 个相干性窗口',
                15 + int(70 * len(frames) / (duration // 10)))
        if seen >= finish:
            break
    if len(frames) != duration // 10 or buffer.shape[1]:
        raise ValueError('EDF不足以覆盖请求的完整网络分析窗口。')
    return frames


def _demo_blocks(end_seconds, cancel):
    """固定种子的合成多导联时序；没有真实患者，也不模拟诊断标签。"""
    rng = np.random.default_rng(DEMO_SEED)
    time_seen = 0
    # 逐块滤波随机驱动源，左右共同成分随时间缓变，供交互验证。
    sos = signal.butter(2, [4, 8], btype='bandpass', fs=FS, output='sos')
    zi = np.zeros((len(sos), 3, 2))
    nodes = network_nodes()
    sides = np.array([n['position'][0] for n in nodes])
    for _ in range(math.ceil(end_seconds / 10)):
        check_cancel(cancel)
        t = (np.arange(WIDTH) + time_seen) / FS
        shared, zi = signal.sosfilt(sos, rng.normal(size=(3, WIDTH)), axis=-1, zi=zi)
        independent = rng.normal(scale=8, size=(18, WIDTH))
        blend = .5 + .45 * np.sin(2 * np.pi * t / 100)
        left = 48 * shared[0] * blend + 22 * shared[2]
        right = 48 * shared[1] * (1 - blend) + 22 * shared[2]
        weights = np.clip((sides + .8) / 1.6, 0, 1)[:, None]
        common = (1 - weights) * left + weights * right
        alpha = (6 + np.arange(18)[:, None] % 4) * np.sin(2 * np.pi * 10 * t + np.arange(18)[:, None] / 3)
        yield independent + common + alpha
        time_seen += WIDTH


def _payload(frames, source):
    return dict(schema_version=1, software_version='V1.0', software_name='头皮脑电癫痫发作预测软件', source=source,
                method=dict(name='Welch magnitude-squared coherence',
                            display_name='Welch 幅值平方相干性',
                            sampling_hz=FS, window_seconds=10, stride_seconds=10,
                            welch_segment_seconds=2, welch_samples=256, overlap_samples=128,
                            taper='periodic Hann', detrend='constant',
                            estimator='mean_f(|mean_segments(conj(FFT_i) * FFT_j)|² / (P_ii * P_jj))',
                            band_average='频带内离散频点等权平均；左闭右开，gamma含45Hz端点',

"""非pickle特征包：强校验后才能训练，患者身份规范化后才能划分。"""
from dataclasses import dataclass
import io
import json
from pathlib import Path
import zipfile
import numpy as np
from .config import feature_contract
from .storage import atomic_bytes, input_file

def canonical_patient(value):
    if isinstance(value,(bytes,np.bytes_)):
        value=value.decode('utf-8',errors='strict')
    value = str(value).strip()
    if value.upper().startswith('CHB-MIT:'):
        value='CHB-MIT:'+value.split(':',1)[1].lower()
    elif value.lower().startswith('chb'):
        value=value.lower()
    if value in {'chb01','chb21','CHB-MIT:chb01','CHB-MIT:chb21','chb01+chb21'}:
        return 'CHB-MIT:chb01+chb21'
    if value.startswith('chb'):
        return 'CHB-MIT:'+value
    return value

@dataclass
class Dataset:
    X: np.ndarray
    y: np.ndarray
    patient: np.ndarray
    split: np.ndarray
    time_seconds: np.ndarray
    record: np.ndarray
    event: np.ndarray
    metadata: dict

    def validate(self):
        x = np.asarray(self.X)
        if x.ndim != 2 or x.shape[1] != 90 or not 0 < len(x) <= 500000:
            raise ValueError('X 必须是1至500000行、90列的特征矩阵。')
        if x.dtype.kind not in 'fi' or not np.isfinite(x).all():
            raise ValueError('特征含NaN、无穷或非数值；请核验数据，不能静默填补。')
        n = len(x)
        for name in ('y','patient','split','time_seconds','record','event'):
            if np.asarray(getattr(self,name)).shape != (n,):
                raise ValueError(f'{name} 长度必须与 X 行数相同。')
        if self.y.dtype.kind not in 'iu' or not np.isin(self.y,[-1,0,1]).all():
            raise ValueError('标签 y 须为整数 -1/0/1，-1 表示未知。')
        for name in ('patient','split','record','event'):
            if getattr(self,name).dtype.kind not in 'US':
                raise ValueError(f'{name} 必须是非对象字符串数组。')
            if getattr(self,name).dtype.kind=='S':
                setattr(self,name,np.array([v.decode('utf-8',errors='strict') for v in getattr(self,name)]))
        self.patient = np.array([canonical_patient(p) for p in self.patient])
        if any(not p or len(p)>128 for p in self.patient) or any(not r for r in self.record):
            raise ValueError('患者或记录标识为空/过长。')
        if not np.isin(self.split,['train','validation','test','inference']).all():
            raise ValueError('split 只能是 train/validation/test/inference。')
        if self.time_seconds.dtype.kind not in 'fi' or not np.isfinite(self.time_seconds).all() or (self.time_seconds<0).any():
            raise ValueError('time_seconds 必须为非负有限数字。')
        memberships = {}
        recording_patients = {}
        coordinates = set()
        for p,s,r,t in zip(self.patient,self.split,self.record,self.time_seconds):
            if p in memberships and memberships[p] != s:
                raise ValueError(f'患者泄漏：{p} 同时位于多个分区。')
            memberships[p] = s
            if r in recording_patients and recording_patients[r]!=p:
                raise ValueError('同一记录被赋给不同患者，可能产生数据泄漏。')
            recording_patients[r]=p
            key = (str(p),str(r),float(t))
            if key in coordinates:
                raise ValueError('存在重复的患者/记录/窗口时刻。')
            coordinates.add(key)
        if any(not str(e).strip() for e in self.event[self.y==1]):
            raise ValueError('阳性窗口必须关联真实事件标识。')
        if self.metadata.get('feature_contract') != feature_contract():
            raise ValueError('特征顺序或信号协议不匹配本软件版本。')
        return self

def load_dataset(path):
    path = input_file(path,'.npz')
    # 压缩包在解压前检查声明大小；禁止对象数组与 pickle。
    if path.stat().st_size > 600*1024*1024:
        raise ValueError('数据包超过600MB大小预算，请分批处理。')
    expected = {'X','y','patient','split','time_seconds','record','event','metadata'}
    with zipfile.ZipFile(path) as archive:
        names = archive.namelist()
        if len(names)!=8 or set(names)!={n+'.npy' for n in expected}:
            raise ValueError('NPZ 内容字段不完整、重复或包含额外项。')
        if sum(i.file_size for i in archive.infolist())>600*1024*1024:
            raise ValueError('NPZ 解压后的大小超过600MB预算。')
    with np.load(path,allow_pickle=False) as archive:
        arrays = {key:archive[key] for key in expected}
    metadata = arrays.pop('metadata')
    if metadata.shape != () or metadata.dtype.kind not in 'US' or len(str(metadata.item()))>4*1024*1024:
        raise ValueError('metadata 须为标量 JSON 字符串。')
    parsed = json.loads(str(metadata.item()))
    if not isinstance(parsed,dict):
        raise ValueError('metadata 必须为JSON对象。')
    return Dataset(**arrays,metadata=parsed).validate()

def save_dataset(path,dataset):
    dataset.validate()
    buffer = io.BytesIO()
    np.savez_compressed(buffer,X=dataset.X,y=dataset.y,patient=dataset.patient,
                        split=dataset.split,time_seconds=dataset.time_seconds,
                        record=dataset.record,event=dataset.event,
                        metadata=np.array(json.dumps(dataset.metadata,ensure_ascii=False,allow_nan=False)))
    atomic_bytes(path,buffer.getvalue())

"""独立患者宏平均与合并窗指标；AP明确区别于梯形PR曲线面积。"""
import csv
import io
import math
import numpy as np
from sklearn.metrics import (average_precision_score, roc_auc_score, f1_score,
                             balanced_accuracy_score, confusion_matrix, roc_curve,
                             precision_recall_curve)
from .storage import atomic_bytes, input_file
from .data import canonical_patient

def one_metrics(y,p,threshold):
    two_classes = len(np.unique(y))==2
    predicted = (p>=threshold).astype(int)
    return dict(n=len(y),positives=int((y==1).sum()),prevalence=float(y.mean()),
                auroc=float(roc_auc_score(y,p)) if two_classes else None,
                ap=float(average_precision_score(y,p)) if two_classes else None,
                macro_f1=float(f1_score(y,predicted,labels=[0,1],average='macro',zero_division=0)),
                balanced_accuracy=float(balanced_accuracy_score(y,predicted)) if two_classes else None,
                confusion=confusion_matrix(y,predicted,labels=[0,1]).tolist(),
                support='both_classes' if two_classes else 'single_class')

def evaluate(y,probability,patient,threshold=.5):
    y,p,patient = np.asarray(y),np.asarray(probability,dtype=float),np.asarray(patient)
    if isinstance(threshold,bool) or not isinstance(threshold,(int,float)) or not math.isfinite(threshold) or not 0<=threshold<=1:
        raise ValueError('阈值必须为0到1的有限数字。')
    if y.ndim!=1 or not len(y) or p.shape!=y.shape or patient.shape!=y.shape:
        raise ValueError('标签、分数、患者标识须为同长度的一维数组。')
    if y.dtype.kind not in 'iu' or not np.isin(y,[-1,0,1]).all():
        raise ValueError('评估标签必须为整数 -1/0/1。')
    if not np.isfinite(p).all() or not ((p>=0)&(p<=1)).all():
        raise ValueError('预测分数含非有限数或超出[0,1]。')
    if patient.dtype.kind not in 'US' or any(not str(v).strip() for v in patient):
        raise ValueError('患者标识须为非空字符串。')
    patient = np.array([canonical_patient(v) for v in patient])
    known = y!=-1
    if not known.any():
        raise ValueError('全部标签未知，无法计算分类效果；仍可导出分数。')
    patients = []
    for identity in sorted(set(patient[known])):
        selected = known&(patient==identity)
        patients.append(dict(patient=str(identity),**one_metrics(y[selected],p[selected],threshold)))
    keys = ('auroc','ap','macro_f1','balanced_accuracy','prevalence')
    macro = {}
    support = {}
    for key in keys:
        values = [row[key] for row in patients if row[key] is not None]
        macro[key] = float(np.mean(values)) if values else None
        support[key] = len(values)
    return dict(macro=macro,pooled=one_metrics(y[known],p[known],threshold),patients=patients,
                macro_support=support,excluded_unknown=int((~known).sum()),threshold=threshold,
                definitions=dict(ap='average_precision_score；不是梯形PR面积',
                                 macro='逐患者等权；缺类AUROC/AP为空并报告支持数',
                                 score='未校准模型分数；不可解释为临床发作概率',
                                 scope='窗口分类评估；不推断事件敏感度或FAR/h'))

def curves(y,p):
    y,p=np.asarray(y),np.asarray(p)
    known=y!=-1
    if len(np.unique(y[known]))!=2:
        return {}
    fpr,tpr,_=roc_curve(y[known],p[known],drop_intermediate=True)
    precision,recall,_=precision_recall_curve(y[known],p[known])
    # 展示点降采样仅影响图形，指标由完整数据计算。
    def points(a,b):
        ids=np.unique(np.linspace(0,len(a)-1,min(300,len(a)),dtype=int))
        return [[float(a[i]),float(b[i])] for i in ids]
    return {'roc':points(fpr,tpr),'pr':points(recall,precision)}

def write_predictions(path,dataset,p,mask=None):
    selected=np.ones(len(p),dtype=bool) if mask is None else mask
    stream=io.StringIO(newline='')
    writer=csv.writer(stream)
    writer.writerow(['patient','record','time_seconds','y','probability'])
    for i in np.flatnonzero(selected):
        writer.writerow([dataset.patient[i],dataset.record[i],float(dataset.time_seconds[i]),int(dataset.y[i]),float(p[i])])
    atomic_bytes(path,stream.getvalue().encode('utf-8-sig'))

def read_predictions(path):
    path=input_file(path,'.csv')
    if path.stat().st_size>150*1024*1024:
        raise ValueError('CSV 超过150MB预算。')
    rows,coordinates=[],set()
    with path.open(encoding='utf-8-sig',newline='') as handle:
        reader=csv.DictReader(handle)
        if reader.fieldnames!=['patient','record','time_seconds','y','probability']:
            raise ValueError('CSV字段或顺序错误，须使用本软件导出格式。')
        for row in reader:
            if len(rows)>=500000 or None in row or any(v is None for v in row.values()):
                raise ValueError('CSV行数或字段异常。')
            identity=canonical_patient(row['patient'])
            t=float(row['time_seconds'])
            key=(identity,row['record'],t)
            if not math.isfinite(t) or t<0 or not identity or not row['record'] or key in coordinates:
                raise ValueError('CSV含重复窗口或非法时间/标识。')
            coordinates.add(key)
            rows.append((identity,row['record'],t,int(row['y']),float(row['probability'])))
    if not rows:
        raise ValueError('CSV没有预测行。')
    return rows

"""只读导入既有研究预测，重新计算窗口指标并保留事件评估来源。"""
from __future__ import annotations

import csv
import io
import json
import math
from pathlib import Path
import re

from .storage import atomic_bytes, check_cancel, file_hash, read_json, write_json


def _number(value, label, *, lower=None, upper=None):
    if isinstance(value, bool) or not isinstance(value, (int, float)) or not math.isfinite(value):
        raise ValueError(f'历史预测的 {label} 必须是有限数值。')
    if lower is not None and value < lower or upper is not None and value > upper:
        raise ValueError(f'历史预测的 {label} 超出合法范围。')
    return float(value)


def _unique_object(pairs):
    value = {}
    for key, item in pairs:
        if key in value:
            raise ValueError(f'历史预测含重复 JSON 键：{key}')
        value[key] = item
    return value


def _reject_constant(value):
    raise ValueError(f'历史预测含非法数值：{value}')


def _metric(values, names):
    for name in names:
        if name in values:
            return values[name]
    raise ValueError('重算指标缺少字段：' + '/'.join(names))


def _verify_metric(actual, expected, label):
    if actual is None or expected is None:
        equal = actual is None and expected is None
    else:
        actual = _number(actual, label)
        expected = _number(expected, label)
        equal = math.isclose(actual, expected, rel_tol=1e-10, abs_tol=1e-12)
    if not equal:
        raise ValueError(f'重算 {label} 与原始研究指标不一致；停止生成完成结果。')


def _verify_fingerprint_pair(actual, expected, label):
    if any(not isinstance(value, str) or re.fullmatch(r'[0-9a-fA-F]{64}', value) is None
           for value in (actual, expected)):
        raise ValueError(f'历史预测和事件评估的 {label} 缺失或不是合法 SHA256。')
    if actual.lower() != expected.lower():
        raise ValueError(f'历史预测和事件评估的 {label} 不一致。')


def _metric_checks(recomputed, original):
    macro = recomputed['macro']
    checks = {}
    for label, aliases, source_key in (
            ('auroc', ('auroc', 'macro_auroc'), 'macro_auroc'),
            ('ap', ('ap', 'auprc', 'macro_ap', 'macro_auprc'), 'macro_auprc')):
        actual = _metric(macro, aliases)
        expected = original[source_key]
        _verify_metric(actual, expected, 'macro ' + label)
        checks[label] = dict(recomputed=actual, source=expected, matches=True)
    patients = recomputed['patients']
    if isinstance(patients, list):
        patients = {row.get('patient', row.get('patient_id')): row for row in patients}
    if not isinstance(patients, dict) or set(patients) != set(original['per_patient']):
        raise ValueError('重算患者清单与历史窗口指标清单不一致。')
    for patient, expected in original['per_patient'].items():
        row = patients[patient]
        for label, aliases, source_key in (
                ('auroc', ('auroc',), 'auroc'),
                ('ap', ('ap', 'auprc'), 'auprc')):
            _verify_metric(_metric(row, aliases), expected[source_key], patient + ' ' + label)
    checks['per_patient_matches'] = True
    return checks


def _formatted(value):
    return '不可计算' if value is None else f'{value:.8f}'


def import_history(project, out, log=None, cancel=None):
    """读取原项目封存测试预测；不打开 EEG、不训练或调整历史模型。

    project 可以指向 EEGProject 或其 prediction_restart 子目录。完成结果包含
    全部未知标签窗口，指标函数自行排除未知标签。输出目录不能与原项目交叠。
    """
    from .evaluation import evaluate

    project = Path(project).expanduser().resolve()
    if (project / 'prediction_restart').is_dir():
        project = project / 'prediction_restart'
    if not project.is_dir():
        raise ValueError('原研究项目目录不存在。')
    out = Path(out).expanduser().resolve()
    if out.is_relative_to(project) or project.is_relative_to(out):
        raise ValueError('历史导入的输出目录不能位于原研究项目内或包含原项目。')
    roots = {

"""仅训练分区拟合归一化；数值JSON模型部署，无pickle反序列化。"""
import warnings
import numpy as np
from scipy.special import expit
from sklearn.linear_model import LogisticRegression

from .config import feature_contract

def load_model(path):
    model=read_json(path)
    if not isinstance(model,dict) or model.get('schema_version')!=1 or model.get('feature_contract')!=feature_contract():
        raise ValueError('模型版本、通道顺序或特征协议不兼容。')
    for key in ('mean','scale','coef'):
        raw=model.get(key)
        if not isinstance(raw,list) or any(isinstance(v,bool) or not isinstance(v,(int,float)) for v in raw):
            raise ValueError(f'模型 {key} 必须是数字数组，不接受字符串或布尔值。')
        arr=np.asarray(raw,dtype=float)
        if arr.shape!=(90,) or not np.isfinite(arr).all():
            raise ValueError(f'模型 {key} 须为90个有限数字。')
        if key=='scale' and not (arr>0).all():
            raise ValueError('模型归一化尺度必须全部大于0。')
    for key in ('intercept','threshold'):
        value=model.get(key)
        if isinstance(value,bool) or not isinstance(value,(int,float)) or not np.isfinite(value):
            raise ValueError(f'模型{key}不是有限数字。')
    if not 0<=model['threshold']<=1:
        raise ValueError('模型阈值越界。')
    for key in ('training_patients','validation_patients'):
        values=model.get(key)
        if not isinstance(values,list) or not all(isinstance(p,str) and p for p in values) or len(set(values))!=len(values):
            raise ValueError('模型患者来源元数据不完整。')
        model[key]=[canonical_patient(p) for p in values]
        if len(set(model[key]))!=len(values):
            raise ValueError('模型包含同一患者的重复别名。')
    if set(model['training_patients'])&set(model['validation_patients']):
        raise ValueError('模型训练与验证患者重叠。')
    return model

def predict(X,model):
    X=np.asarray(X,dtype=float)
    if X.ndim!=2 or X.shape[1]!=90 or not np.isfinite(X).all():
        raise ValueError('推理特征须为有限的(N,90)矩阵。')
    with np.errstate(over='raise',invalid='raise'):
        score=((X-np.asarray(model['mean']))/np.asarray(model['scale']))@np.asarray(model['coef'])+model['intercept']
    return expit(score)

def train(dataset,out,log,cancel):
    dataset.validate()
    subsets={s:(dataset.split==s)&(dataset.y!=-1) for s in ('train','validation','test')}
    for s in ('train','validation','test'):
        if not subsets[s].any():
            raise ValueError(f'{s} 分区没有可评价样本，请按患者分配并核验标注。')
    selected=subsets['train']
    if len(np.unique(dataset.y[selected]))!=2:
        raise ValueError('训练分区缺少一类，不能训练二分类模型。')
    check_cancel(cancel)
    # 固定种子、每患者每类最多200窗；事件轮流取样，避免长事件垄断预算。
    rng=np.random.default_rng(20260929)
    chosen=[]
    for identity in sorted(set(dataset.patient[selected])):
        for label in (0,1):
            indices=np.flatnonzero(selected&(dataset.patient==identity)&(dataset.y==label))
            groups={e:list(rng.permutation(indices[dataset.event[indices]==e])) for e in sorted(set(dataset.event[indices]))}
            count=0
            while any(groups.values()) and count<200:
                for bucket in groups.values():
                    if bucket and count<200:
                        chosen.append(bucket.pop()); count+=1
    ids=np.asarray(chosen,dtype=int)
    weights=np.zeros(len(ids))
    patients=dataset.patient[ids]
    labels=dataset.y[ids]
    events=dataset.event[ids]
    for identity in set(patients):
        classes=set(labels[patients==identity])
        for label in classes:
            here=(patients==identity)&(labels==label)
            names=set(events[here]) if label==1 else {''}
            for event in names:
                mask=here&(events==event) if label==1 else here
                weights[mask]=1/(len(set(patients))*len(classes)*len(names)*mask.sum())
    weights/=weights.mean()
    log(f'拟合训练患者归一化与逻辑回归：{len(set(patients))}名患者，{len(ids)}个窗。',30)
    scaler=StandardScaler().fit(dataset.X[ids],sample_weight=weights)
    classifier=LogisticRegression(C=1.0,max_iter=1000,solver='lbfgs',random_state=20260929)
    with warnings.catch_warnings(record=True) as caught:
        warnings.simplefilter('always',ConvergenceWarning)
        classifier.fit(scaler.transform(dataset.X[ids]),labels,sample_weight=weights)
    check_cancel(cancel)
    if any(issubclass(w.category,ConvergenceWarning) for w in caught):
        raise RuntimeError('逻辑回归未收敛；任务失败，不能将该模型当作有效结果。')
    model=dict(schema_version=1,feature_contract=feature_contract(),mean=scaler.mean_.tolist(),
               scale=scaler.scale_.tolist(),coef=classifier.coef_[0].tolist(),intercept=float(classifier.intercept_[0]),
               threshold=.5,training_patients=sorted(set(patients)),
               validation_patients=sorted(set(dataset.patient[subsets['validation']])),
               score_semantics='uncalibrated',provenance=dict(kind=dataset.metadata.get('kind','user_data'),
               seed=20260929,training_windows=len(ids),threshold_policy='fixed_0.5_no_test_tuning'))
    write_json(out/'model.json',model)
    log('模型已冻结；使用固定0.5阈值评估独立测试患者。',70)
    return model

def waveform_preview(reader,record_id,identity,channels=8):
    """从 EDF 中间取 10 秒因果滤波波形用于界面预览，限制总点数避免任务记录膨胀。"""
    start = 30.0 if reader.duration >= 40 else max(0.0, reader.duration - 10)
    preview = reader.segment(start,10,channels,max_points=1280)
    preview['record']=record_id
    preview['patient']=identity
    return preview

def prepare(manifest,out,log,cancel):
    path=input_file(manifest,'.json')
    config=read_json(path)
    if not isinstance(config,dict) or config.get('schema_version')!=1:
        raise ValueError('清单需要 schema_version=1。')
    records=config.get('records')
    if not isinstance(records,list) or not 1<=len(records)<=5000:
        raise ValueError('清单 records 须为1至5000条记录。')
    source_root=Path(config.get('source_root',path.parent)).resolve()
    features,labels,patients,splits,times,identities,event_ids=[],[],[],[],[],[],[]
    waveform_previews=[]
    excluded=Counter()
    sources=[]
    seen=set()
    assignments={}
    for index,record in enumerate(records):
        check_cancel(cancel)
        if not isinstance(record,dict) or record.get('annotations_complete') is not True:
            raise ValueError('每条记录必须明确 annotations_complete=true；未核验标注不能生成阴性。')
        source=(source_root/record.get('path','')).resolve()
        if not source.is_relative_to(source_root):
            raise ValueError('EDF路径超出 source_root。')
        source=input_file(source,'.edf')
        if source in seen:
            raise ValueError('同一EDF在清单中重复出现。')
        seen.add(source)
        identity=canonical_patient(record.get('patient',''))
        split=record.get('split')
        if not identity or split not in {'train','validation','test','inference'}:
            raise ValueError('清单患者标识为空或split非法。')
        if identity in assignments and assignments[identity]!=split:
            raise ValueError('同一患者不能跨分区，chb01与chb21亦视为同一患者。')
        assignments[identity]=split
        log(f'读取 {index+1}/{len(records)}：{source.name}',int(5+80*index/len(records)))
        reader=EDFReader(source)
        if 'seizures' not in record or not isinstance(record['seizures'],list):
            raise ValueError('清单必须显式填写 seizures 数组；空数组代表已核验无发作。')
        events=validate_events(record['seizures'],reader.duration)
        if reader.duration*128>128*86400:
            raise ValueError('单条记录超过24小时，请核验头信息并分段处理。')
        record_id=source.relative_to(source_root).as_posix()
        sources.append(dict(path=str(source),sha256=file_hash(source),duration=reader.duration,
                            record=record_id,patient=identity))
        if len(waveform_previews)<4:
            waveform_previews.append(waveform_preview(reader,record_id,identity))
        causal=CausalFilter(reader.fs)
        buffer=np.empty((18,0),dtype=np.float32)
        origin,total=0,0
        width=1280
        next_end=5120
        record_id=source.relative_to(source_root).as_posix()
        for block in reader.blocks():
            check_cancel(cancel)
            # 与原项目一致：导联先量化为float32伏，再因果滤波，缓存为float32微伏。
            volts=(block*1e-6).astype(np.float32).astype(np.float64)
            converted=(causal.transform(volts)*1e6).astype(np.float32)
            total+=converted.shape[1]
            buffer=np.concatenate((buffer,converted),axis=1)
            while next_end<=total:
                if len(features)>=500000:
                    raise ValueError('特征包超过500000窗口预算，请分批准备数据。')
                window=buffer[:,next_end-width-origin:next_end-origin]
                t=next_end/128
                y,reason,event=label_window(t,reader.duration,events)
                features.append(extract_features(window))
                labels.append(y); patients.append(identity); splits.append(split)
                times.append(t); identities.append(record_id)
                event_ids.append('' if event is None else f'{record_id}:event:{event}')
                if y==-1:
                    excluded[reason]+=1
                next_end+=width
            discard=max(0,min(buffer.shape[1],next_end-width-origin))
            buffer=buffer[:,discard:]
            origin+=discard
        if file_hash(source)!=sources[-1]['sha256']:
            raise ValueError('源EDF读取期间改变，任务失败。')
    if not features:
        raise ValueError('没有完整10秒窗口；文件需要超过40秒。')
    dataset=Dataset(np.asarray(features,dtype=np.float32),np.asarray(labels,dtype=np.int8),
                    np.array(patients),np.array(splits),np.array(times),np.array(identities),
                    np.array(event_ids),dict(feature_contract=feature_contract(),kind='user_edf',
                    sources=sources,manifest_sha256=file_hash(path),cross_file_join=False,
                    annotation_semantics='user_confirmed_complete_file_inventory')).validate()
    check_cancel(cancel)
    save_dataset(out/'dataset.npz',dataset)
    summary=dataset_summary(dataset)

def run_action(action,params,out,log,cancel):
    out=Path(out); out.mkdir(parents=True,exist_ok=True)
    file_keys={'prepare':['manifest'],'train':['dataset'],'predict':['dataset','model'],
               'evaluate':['predictions']}.get(action,[])
    input_files={key:dict(path=str(input_file(params.get(key))),sha256=file_hash(input_file(params.get(key))))
                 for key in file_keys}

        result=prepare(params.get('manifest'),out,log,cancel)
    elif action in {'train','demo'}:

        result=training_result(dataset,out,log,cancel)

        model=load_model(params.get('model'))

            raise ValueError('缺少模型训练患者来源，无法核验独立测试评估；请补齐来源或使用 inference 分区。')
        declared=set(model['training_patients'])|set(model['validation_patients'])
        if set(dataset.patient[dataset.split=='test'])&declared:
            raise ValueError('独立测试患者出现在模型训练或验证来源中，拒绝泄漏评估。')
        log('校验模型协议并生成全部窗口的未校准风险分数。',30)
        p=predict(dataset.X,model)
        write_predictions(out/'predictions.csv',dataset,p)
        metrics=None
        mask=dataset.split=='test'
        if ((dataset.y!=-1)&mask).any():
            metrics=evaluate(dataset.y[mask],p[mask],dataset.patient[mask],model['threshold'])
            metrics['source_kind']=dataset.metadata.get('kind','user_data')
            metrics['curves']=curves(dataset.y[mask],p[mask])
            write_json(out/'metrics.json',metrics); write_report(out,metrics,metrics['source_kind'])
        # 单条记录预览，不能连接不同文件的相对秒数。
        first=dataset.record[mask][0] if mask.any() else dataset.record[0]

        previews=[]

        # 旧数据包未记 record，只有唯一文件名匹配时才允许展示，避免混淆患者。
        if not matching:
            matching=[s for s in sources if Path(s.get('path','')).name==Path(str(first)).name]
        if len(matching)==1:
            try:
                source=matching[0]
                if file_hash(source['path'])!=source.get('sha256'):
                    raise ValueError('波形源文件已改变。')
                previews.append(waveform_preview(EDFReader(input_file(source['path'],'.edf')),
                                                 str(first),first_patient))
            except (ValueError,OSError):
                previews=[]
        result=dict(summary='全部窗口分数已导出；未知标签不参与分类评估。',metrics=metrics,
                    preview=[dict(patient=str(dataset.patient[i]),record=str(first),time_seconds=float(dataset.time_seconds[i]),
                                  y=int(dataset.y[i]),probability=float(p[i])) for i in ids],
                    waveform_previews=previews,
                    artifacts=[artifact('predictions.csv','窗口预测')],output_paths=dict(predictions=str(out/'predictions.csv'),
                    dataset=str(input_file(params['dataset'])),model=str(input_file(params['model']))))
        if metrics:
            result['artifacts'] += [artifact('metrics.json','独立测试指标'),artifact('report.txt','效果报告')]
    elif action=='evaluate':
        rows=read_predictions(params.get('predictions'))
        y=np.array([r[3] for r in rows]); p=np.array([r[4] for r in rows]); patients=np.array([r[0] for r in rows])
        metrics=evaluate(y,p,patients,params.get('threshold',.5))
        metrics['source_kind']='external_csv_descriptive'
        metrics['curves']=curves(y,p)
        write_json(out/'metrics.json',metrics); write_report(out,metrics,'CSV描述性分析，独立性由数据来源决定')
        result=dict(summary='CSV分析完成；来源独立性不能由CSV本身证明。',metrics=metrics,
                    artifacts=[artifact('metrics.json','评估指标'),artifact('report.txt','分析报告')])
    elif action=='history':
        from .history import import_history

    elif action=='connectivity':
        from .connectivity import run_connectivity
        result=run_connectivity(params,out,log,cancel)
    else:
        raise ValueError('不支持的任务类型。')
    check_cancel(cancel)
    metrics=result.get('metrics')
    if metrics:
        warnings=[]
        chart_names=make_charts(metrics,out,warnings)
        if warnings:
            result.setdefault('warnings',[]).extend(warnings)
            for warning in warnings:
                log(warning)
        if chart_names:
            labels={'roc_curve.png':'ROC曲线图','pr_curve.png':'PR曲线图',
                    'patient_metrics.png':'患者指标对比图','confusion_matrix.png':'混淆矩阵图'}
            result['artifacts'] += [artifact(name,labels[name]) for name in chart_names]
    check_cancel(cancel)
    input_files.update(result.pop('input_files',{}))
    if any(file_hash(item['path'])!=item['sha256'] for item in input_files.values()):
        raise ValueError('输入文件在任务过程中改变；拒绝将结果标为完成。')
    check_cancel(cancel)
    write_json(out/'provenance.json',dict(action=action,params=params,protocol=PROTOCOL.to_dict(),
               feature_contract=feature_contract(),python=platform.python_version(),

class JobManager:
    def __init__(self,root):
        self.root=Path(root).resolve(); self.outputs=self.root/'outputs'
        self.outputs.mkdir(parents=True,exist_ok=True)
        self.lock=threading.RLock(); self.jobs={}; self.cancel_events={}
        states=sorted(self.outputs.glob('*/job.json'),key=lambda p:p.stat().st_mtime,reverse=True)[:40]
        for path in states:
            try:
                job=read_json(path,limit=8*1024*1024)
                if not self.valid_saved_job(job,path.parent.name):
                    continue
                if job['status'] in {'queued','running'}:
                    job['status']='failed'; job['error']='上次程序退出时任务未完成，请重新提交。'
                self.jobs[job['id']]=job
            except (ValueError,OSError,KeyError):
                continue

    @staticmethod
    def valid_saved_job(job,id):
        if not isinstance(job,dict) or job.get('id')!=id:
            return False
        if job.get('action') not in ('demo','prepare','train','predict','evaluate','history','connectivity') or job.get('status') not in ('queued','running','completed','failed','cancelled'):
            return False
        for key in ('created','progress'):
            value=job.get(key)
            if isinstance(value,bool) or not isinstance(value,(int,float)) or (isinstance(value,float) and not math.isfinite(value)):
                return False
        if not 0<=job['progress']<=100 or not isinstance(job.get('logs'),list) or any(not isinstance(x,str) for x in job['logs']):
            return False
        result=job.get('result')
        if result is not None and not isinstance(result,dict):
            return False
        if result is not None:
            artifacts=result.get('artifacts',[])
            if not isinstance(artifacts,list) or any(not isinstance(a,dict) or not isinstance(a.get('name'),str) for a in artifacts):
                return False
            paths=result.get('output_paths',{})
            if not isinstance(paths,dict) or any(not isinstance(v,str) for v in paths.values()):
                return False
        return job.get('error') is None or isinstance(job.get('error'),str)

    def snapshot(self,id=None):
        with self.lock:
            value=self.jobs.get(id) if id else dict(jobs=sorted(self.jobs.values(),key=lambda j:j['created'],reverse=True)[:40])
            return json.loads(json.dumps(value,ensure_ascii=False,allow_nan=False))

    def save(self,job):
        write_json(self.outputs/job['id']/'job.json',job)

    def submit(self,action,params):
        if action not in {'demo','prepare','train','predict','evaluate','history','connectivity'} or not isinstance(params,dict):
            raise ValueError('任务类型或参数格式不合法。')
        with self.lock:
            if any(j['status'] in {'queued','running'} for j in self.jobs.values()):
                raise RuntimeError('已有任务运行，请等待完成或取消。')
            id=uuid4().hex[:16]
            job=dict(id=id,action=action,status='queued',progress=0,logs=[],created=time.time(),result=None,error=None)
            self.jobs[id]=job; cancel=threading.Event(); self.cancel_events[id]=cancel
            self.save(job)
            threading.Thread(target=self.execute,args=(job,params,cancel),daemon=True).start()
            return self.snapshot(id)

    def execute(self,job,params,cancel):
        out=self.outputs/job['id']
        def log(message,progress=None):
            with self.lock:
                job['logs'].append(time.strftime('%H:%M:%S')+' '+str(message))
                job['logs']=job['logs'][-200:]
                if progress is not None:
                    job['progress']=int(progress)
                self.save(job)
        with self.lock:
            job['status']='running'; self.save(job)
        try:
            result=run_action(job['action'],params,out,log,cancel)
            with self.lock:
                check_cancel(cancel)
                job.update(status='completed',progress=100,result=result)
        except Cancelled as exc:
            with self.lock:
                job.update(status='cancelled',error=str(exc))
        except Exception as exc:
            with self.lock:
                job.update(status='failed',error=f'{type(exc).__name__}: {exc}')
            (out/'error.log').write_text(traceback.format_exc(),encoding='utf-8')
        finally:
            with self.lock:
                job['finished']=time.time(); self.save(job)
                self.cancel_events.pop(job['id'],None)

    def cancel(self,id):
        with self.lock:
            if id not in self.cancel_events or self.jobs[id]['status'] not in {'queued','running'}:
                raise ValueError('该任务已结束或不存在。')
            self.cancel_events[id].set()
            return dict(message='已请求取消，待当前计算步骤结束后停止。')

    def defaults(self):
        values=dict(manifest=str(self.root/'examples'/'manifest.example.json'),
                    model=str(self.root/'examples'/'historical_model.json'))
        try:
            manifest=read_json(values['manifest'])
            first=manifest['records'][0]
            candidate=(Path(manifest.get('source_root',self.root/'examples'))/first['path']).resolve()
            if candidate.is_file() and candidate.suffix.lower()=='.edf':
                values['edf_path']=str(candidate)
        except (ValueError,OSError,KeyError,IndexError,TypeError):
            pass
        with self.lock:
            for job in sorted(self.jobs.values(),key=lambda j:j['created']):
                if job['status']=='completed':
                    paths=(job.get('result') or {}).get('output_paths',{})
                    values.update((k,v) for k,v in paths.items() if k not in {'dataset','model'})
                    if paths.get('dataset'):
                        values['dataset']=paths['dataset']
                        # 同一次任务的数据与模型成对复用，避免真实数据默认配上合成模型。
                        values['model']=paths.get('model',str(self.root/'examples'/'historical_model.json'))
        return values

class Handler(BaseHTTPRequestHandler):
    server_version='LocalEEGResearch/1.0'
    def log_message(self,*args):
        pass

    def valid_host(self):
        port=self.server.server_port
        return self.headers.get('Host') in {f'127.0.0.1:{port}',f'localhost:{port}'}

    def respond(self,status,payload,content_type='application/json; charset=utf-8',filename=None):
        if not isinstance(payload,bytes):
            payload=json.dumps(payload,ensure_ascii=False,allow_nan=False).encode('utf-8')
        self.send_response(status)
        self.send_header('Content-Type',content_type)
        self.send_header('Content-Length',str(len(payload)))
        self.send_header('Cache-Control','no-store')
        self.send_header('X-Content-Type-Options','nosniff')
        self.send_header('Referrer-Policy','no-referrer')
        self.send_header('Content-Security-Policy',"default-src 'self'; script-src 'self'; style-src 'self'; img-src 'self' data:; connect-src 'self'; frame-ancestors 'none'; base-uri 'none'; form-action 'self'")
        if filename:
            self.send_header('Content-Disposition',f'attachment; filename="{filename}"')
        self.end_headers()
        try:
            self.wfile.write(payload)
        except (ConnectionError,OSError):
            pass

    def do_GET(self):
        if not self.valid_host():
            return self.respond(403,dict(error='仅允许本机地址访问。'))
        try:
            url=urlsplit(self.path); query=parse_qs(url.query)
            manager=self.server.manager
            if url.path=='/api/info':
                return self.respond(200,dict(app_name='头皮脑电癫痫发作预测软件',version=__version__,
                    root=str(manager.root),project_path='research_results',
                    python_version=platform.python_version(),protocol=PROTOCOL.to_dict(),
                    csrf_token=self.server.csrf,default_paths=manager.defaults(),history=None))
            if url.path=='/api/jobs':
                result=manager.snapshot(query.get('id',[None])[0])
                return self.respond(200 if result is not None else 404,result or dict(error='任务不存在。'))
            if url.path=='/api/network-demo':
                return self.respond(200,(manager.root/'examples'/'network_demo.json').read_bytes())
            if url.path in {'/api/download','/api/artifact'}:
                id=query.get('job',[''])[0]; name=query.get('name',[''])[0]
                job=manager.snapshot(id)
                allowed={a['name'] for a in (job or {}).get('result',{}).get('artifacts',[])} if job and job.get('result') else set()
                if not job or job['status']!='completed' or name not in allowed:
                    return self.respond(404,dict(error='未完成任务的文件或文件名不存在。'))
                directory=(manager.outputs/id).resolve(); path=(directory/name).resolve()
                if not path.is_relative_to(directory) or not path.is_file() or path.name!=name:
                    return self.respond(403,dict(error='下载路径不合法。'))
                if url.path=='/api/artifact':
                    # 仅允许不会执行主动内容的可视化产物内联显示。
                    mime={'.png':'image/png','.json':'application/json; charset=utf-8'}.get(path.suffix.lower())
                    if mime is None:
                        return self.respond(403,dict(error='该产物仅支持下载。'))
                    return self.respond(200,path.read_bytes(),mime)
                return self.respond(200,path.read_bytes(),'application/octet-stream',name)
            static={'/':'index.html','/index.html':'index.html','/app.js':'app.js','/style.css':'style.css'}
            for name in ('brain-network.js','brain-network.css','vendor/three.module.js',
                         'vendor/three.core.js','vendor/OrbitControls.js'):
                static['/'+name]=name
            if url.path not in static:
                return self.respond(404,dict(error='页面不存在。'))
            path=manager.root/'web'/static[url.path]
            mime={'.js':'text/javascript','.css':'text/css','.html':'text/html'}.get(path.suffix,'text/plain')
            return self.respond(200,path.read_bytes(),mime+'; charset=utf-8')
        except (ValueError,OSError,KeyError) as exc:
            return self.respond(400,dict(error=str(exc)))

    def do_POST(self):
        if not self.valid_host():
            return self.respond(403,dict(error='仅允许本机地址访问。'))
        origin=self.headers.get('Origin')
        if origin and origin not in {f'http://127.0.0.1:{self.server.server_port}',f'http://localhost:{self.server.server_port}'}:
            return self.respond(403,dict(error='请求来源不合法。'))
        if not hmac.compare_digest(self.headers.get('X-CSRF-Token',''),self.server.csrf):
            return self.respond(403,dict(error='操作校验失败，请刷新界面重试。'))
        try:
            length=int(self.headers.get('Content-Length','0'))
            if not 0<length<=16384 or self.headers.get('Content-Type','').split(';')[0]!='application/json':
                return self.respond(400,dict(error='请求格式或大小不合法。'))
            self.connection.settimeout(10)
            data=json.loads(self.rfile.read(length))
            if not isinstance(data,dict):
                raise ValueError('请求须为JSON对象。')
            if self.path=='/api/jobs':
                return self.respond(202,self.server.manager.submit(data.get('action'),data.get('params',{})))
            if self.path=='/api/cancel':
                return self.respond(200,self.server.manager.cancel(data.get('id')))
            return self.respond(404,dict(error='接口不存在。'))
        except RuntimeError as exc:
            return self.respond(409,dict(error=str(exc)))
        except (ValueError,OSError,TypeError) as exc:
            return self.respond(400,dict(error=str(exc)))

"""有界内存的经典 EDF 读取、因果预处理、频带特征与删失标签。

不推断文件之间连续性。EDF+、BDF 和不完整标注必须由专门工具核验，不能
静默当作连续经典 EDF。所有电位校准为微伏，输出点保持原始可用时刻。
"""
import math
from pathlib import Path
import re
import numpy as np
from scipy import signal
from .config import BANDS, CHANNELS, PROTOCOL

def normalize_channel(label):
    label = re.sub(r'\s+', '', label.upper()).removeprefix('EEG')
    parts = label.split('-')
    aliases = {'T3': 'T7', 'T4': 'T8', 'T5': 'P7', 'T6': 'P8'}
    return '-'.join(aliases.get(p, p) for p in parts)

class EDFReader:
    """只接受固定长度记录、合法物理单位、等采样率的18双极导联。

    对同名/反向重复导联逐块核验量化误差，冲突即失败。拒绝缺导联、截断、
    多余字节、非有限校准以及无法验证连续性的 EDF+，避免猜测数据含义。
    """
    def __init__(self, path):
        self.path = Path(path)
        with self.path.open('rb') as handle:
            fixed = handle.read(256)
            if len(fixed) != 256 or fixed[:8].strip() != b'0':
                raise ValueError('不是支持的经典 EDF 文件。')
            def number(a, b, cast=float):
                try:
                    value = cast(fixed[a:b].decode('ascii').strip())
                except (ValueError, UnicodeError) as exc:
                    raise ValueError('EDF 固定头字段不合法。') from exc
                if not math.isfinite(value):
                    raise ValueError('EDF 头字段不是有限数。')
                return value
            self.header_bytes = number(184, 192, int)
            if fixed[192:236].strip():
                raise ValueError('只支持经典 EDF；EDF+ 的时间连续性需单独审计。')
            self.n_records = number(236, 244, int)
            self.record_seconds = number(244, 252)
            count = number(252, 256, int)
            if not 1 <= count <= 256 or self.header_bytes != 256*(1+count):
                raise ValueError('EDF 通道数量或头长度异常。')
            if self.n_records <= 0 or self.record_seconds <= 0:
                raise ValueError('EDF 必须声明正数的记录数量与时长。')
            raw = handle.read(256*count)
            if len(raw) != 256*count:
                raise ValueError('EDF 信号头被截断。')
        fields, offset = {}, 0
        for name, width in [('label',16),('transducer',80),('unit',8),('pmin',8),
                            ('pmax',8),('dmin',8),('dmax',8),('prefilter',80),
                            ('spr',8),('reserved',32)]:
            fields[name] = [raw[offset+i*width:offset+(i+1)*width].decode('ascii').strip()
                            for i in range(count)]
            offset += count*width
        self.channels = []
        for i in range(count):
            channel = {key: fields[key][i] for key in fields}
            for key in ('pmin','pmax','dmin','dmax'):
                channel[key] = float(channel[key])
                if not math.isfinite(channel[key]):
                    raise ValueError('EDF 校准含非有限数。')
            channel['spr'] = int(channel['spr'])
            if channel['spr'] <= 0:
                raise ValueError('EDF 每记录采样点数必须为正。')
            self.channels.append(channel)
        self.offsets = np.cumsum([0]+[c['spr'] for c in self.channels])
        self.record_bytes = int(self.offsets[-1])*2
        if self.record_bytes > 16*1024*1024:
            raise ValueError('单个 EDF 记录超过16MB的内存预算。')
        if self.path.stat().st_size != self.header_bytes+self.record_bytes*self.n_records:
            raise ValueError('EDF 长度与头声明不一致（截断或含多余数据）。')
        labels = [normalize_channel(c['label']) for c in self.channels]
        self.plan = []
        for pair in CHANNELS:
            a, b = pair.split('-')
            options = [(i, 1) for i, lab in enumerate(labels) if lab == pair]
            options += [(i, -1) for i, lab in enumerate(labels) if lab == b+'-'+a]
            if not options:
                raise ValueError(f'缺少双极导联：{pair}；本版不猜测参考方式。')
            self.plan.append(options)
        selected = {i for options in self.plan for i, _ in options}
        rates = [self.channels[i]['spr']/self.record_seconds for i in selected]
        self.fs = rates[0]
        if any(not math.isclose(r, self.fs, abs_tol=1e-9) for r in rates):
            raise ValueError('所需通道的采样率不一致。')
        ratio = self.fs/PROTOCOL.sampling_hz
        if ratio < 1 or not math.isclose(ratio, round(ratio), abs_tol=1e-9):
            raise ValueError('只支持128Hz及其整数倍采样率的因果降采样。')
        self.factor = round(ratio)
        for i in selected:
            c = self.channels[i]
            unit = c['unit'].lower().replace('µ','u').replace('μ','u')
            if unit not in {'v','mv','uv','nv'}:
                raise ValueError(f'不支持的电位单位：{unit}')
            if not c['pmax'] > c['pmin'] or not c['dmax'] > c['dmin']:
                raise ValueError('EDF 校准上下限异常。')
            c['lsb'] = (c['pmax']-c['pmin'])/(c['dmax']-c['dmin'])*{'v':1e6,'mv':1e3,'uv':1,'nv':.001}[unit]
            c['base'] = c['pmin']*{'v':1e6,'mv':1e3,'uv':1,'nv':.001}[unit]
        self.duration = self.n_records*self.record_seconds
        self.selected = selected

    def blocks(self):
        count = max(1, min(int(60/self.record_seconds), (16*1024*1024)//self.record_bytes))
        before = self.path.stat()
        with self.path.open('rb') as handle:
            handle.seek(self.header_bytes)
            for first in range(0, self.n_records, count):
                n = min(count, self.n_records-first)
                raw = handle.read(n*self.record_bytes)
                if len(raw) != n*self.record_bytes:
                    raise ValueError('EDF 在读取过程中被截断。')
                digital = np.frombuffer(raw, dtype='<i2').reshape(n, -1)
                physical = {}
                for i in self.selected:
                    c = self.channels[i]
                    values = digital[:,self.offsets[i]:self.offsets[i+1]].reshape(-1)
                    physical[i] = (values.astype(float)-c['dmin'])*c['lsb']+c['base']
                output = []
                for options in self.plan:
                    index, sign = options[0]
                    base = physical[index]*sign
                    for other, polarity in options[1:]:
                        tolerance = .5*(self.channels[index]['lsb']+self.channels[other]['lsb'])+1e-9
                        if np.max(np.abs(base-physical[other]*polarity)) > tolerance:
                            raise ValueError('重复/反向导联存在冲突，需核验原始数据。')
                    output.append(base)
                yield np.stack(output)
        after = self.path.stat()
        if (before.st_size, before.st_mtime_ns) != (after.st_size, after.st_mtime_ns):
            raise ValueError('EDF 在任务过程中发生变化，请重新运行。')

    def segment(self, start_second, seconds, channels=8, max_points=400):
        """流式保留文件起点的因果状态，仅缓存请求区间；片段最多60秒。

        输出为微伏。展示点抽稀不用于重新分析；采样坐标与原文件一致。
        与特征管线采用相同的float32伏量化与滤波顺序。
        """
        from contextlib import closing
        values = (start_second, seconds, channels, max_points)
        if any(isinstance(v, bool) or not isinstance(v, (int, float)) or not math.isfinite(v) for v in values):
            raise ValueError('波形片段参数须为有限数字。')
        if not 0 <= start_second < self.duration or not 0 < seconds <= 60 or start_second + seconds > self.duration:
            raise ValueError('波形区间须在文件内，且长度不超过60秒。')
        if not 1 <= channels <= 18 or int(channels) != channels or not 50 <= max_points <= 2000 or int(max_points) != max_points:
            raise ValueError('波形导联数或显示点数超出范围。')
        channels, max_points = int(channels), int(max_points)
        fs = PROTOCOL.sampling_hz
        lo, hi = round(start_second * fs), round((start_second + seconds) * fs)
        if hi <= lo:
            raise ValueError('请求片段不足一个采样点。')
        before = self.path.stat()
        causal, cursor, pieces = CausalFilter(self.fs), 0, []
        with closing(self.blocks()) as blocks:
            for block in blocks:
                volts = (block * 1e-6).astype(np.float32).astype(np.float64)
                filtered = (causal.transform(volts) * 1e6).astype(np.float32)
                end = cursor + filtered.shape[1]
                if end > lo and cursor < hi:
                    pieces.append(filtered[:channels, max(0, lo-cursor):min(hi-cursor, filtered.shape[1])].copy())
                cursor = end
                if cursor >= hi:
                    break
        after = self.path.stat()
        if (before.st_size, before.st_mtime_ns) != (after.st_size, after.st_mtime_ns):
            raise ValueError('EDF在波形读取期间改变。')
        clipped = np.concatenate(pieces, axis=1) if pieces else np.empty((channels, 0))
        if clipped.shape[1] != hi-lo:
            raise ValueError('波形区间不完整。')
        step = max(1, int(math.ceil(clipped.shape[1] / max_points)))
        clipped = clipped[:, ::step]
        return dict(start_second=lo/fs, seconds=(hi-lo)/fs, fs=int(fs),
                    display_sampling_hz=fs/step, decimation_note='展示抽稀，分析仍使用128Hz原采样',
                    channels=list(CHANNELS[:channels]),
                    t=(np.arange(clipped.shape[1])*step/fs).tolist(), samples=clipped.tolist())

class CausalFilter:
    """跨块保留 IIR 状态和降采样相位；不用双向滤波与未来样本。"""
    def __init__(self, native_fs):
        ratio = native_fs/PROTOCOL.sampling_hz
        if not math.isfinite(native_fs) or ratio < 1 or not math.isclose(ratio, round(ratio), abs_tol=1e-9):
            raise ValueError('采样率需为128Hz的正整数倍。')
        parts = [signal.butter(4,.5,btype='highpass',fs=native_fs,output='sos'),
                 signal.butter(4,45,btype='lowpass',fs=native_fs,output='sos'),
                 signal.tf2sos(*signal.iirnotch(60,30,fs=native_fs))]
        if native_fs > 128:
            order, cutoff = signal.ellipord(45,64,.2,60,fs=native_fs)
            parts.append(signal.ellip(order,.2,60,cutoff,fs=native_fs,output='sos'))
        self.sos = np.concatenate(parts)
        self.state = np.zeros((len(self.sos),18,2))
        self.factor, self.seen = round(ratio), 0

    def transform(self, x):
        x = np.asarray(x, dtype=float)
        if x.ndim != 2 or x.shape[0] != 18 or not x.shape[1] or not np.isfinite(x).all():
            raise ValueError('信号须为有限的(18, samples)微伏矩阵。')
        filtered, self.state = signal.sosfilt(self.sos,x,axis=-1,zi=self.state)
        first = (-self.seen)%self.factor
        self.seen += x.shape[-1]
        return filtered[:,first::self.factor]

def extract_features(signal_uv, fs=128):
    x = np.asarray(signal_uv)
    if x.shape != (18,1280) or fs != 128 or not np.isfinite(x).all():
        raise ValueError('Welch 输入须为128Hz、10秒、18导联的有限微伏信号。')
    freq, psd = signal.welch(x,fs=fs,window='hann',nperseg=256,noverlap=128,
                             detrend='constant',scaling='density',axis=-1)
    columns = []
    for lo, hi in BANDS:
        grid = np.r_[lo,freq[(freq>lo)&(freq<hi)],hi]
        values = np.stack([np.interp(grid,freq,p) for p in psd])
        columns.append(np.log10(np.trapezoid(values,grid,axis=-1)+1e-12))
    return np.stack(columns,axis=-1).reshape(90).astype(np.float32)

def validate_events(events, duration):
    pairs = []
    for pair in events:
        if not isinstance(pair,(list,tuple)) or len(pair) != 2:
            raise ValueError('发作列表须为 [[起始秒,结束秒], ...]。')
        a,b = pair
        if any(isinstance(v,bool) or not isinstance(v,(int,float)) or not math.isfinite(v) for v in pair):
            raise ValueError('发作时间须为有限数字。')
        if not 0 <= a < b <= duration:
            raise ValueError('发作时间超出文件或起止顺序错误。')
        pairs.append((float(a),float(b)))
    pairs.sort()
    if any(pairs[i][0] < pairs[i-1][1] for i in range(1,len(pairs))):
        raise ValueError('发作事件重叠，需核验标注。')
    return pairs

def label_window(end_seconds, duration, events, protocol=PROTOCOL):
    """返回(y,reason,event_index)。正负类共同要求完整35分钟未来。

    发作需有完整历史窗口且不与此前发作后期重叠，符合原 r1 事件资格。
    标签0仅代表录制条件的未来阴性，文件开始前病史仍未知。
    """
    t, start = end_seconds,end_seconds-protocol.window_seconds
    if not math.isfinite(t+duration) or t < protocol.warmup_seconds+protocol.window_seconds or t > duration:
        return -1,'invalid_window',None
    if any(start<b+protocol.postictal_seconds and a<t for a,b in events):
        return -1,'ictal_or_known_postictal',None
    if any(t<=a<t+protocol.sph_seconds for a,b in events):
        return -1,'within_sph',None
    if t+protocol.horizon > duration:
        return -1,'right_censored',None
    matched = [(i,a,b) for i,(a,b) in enumerate(events) if t+protocol.sph_seconds<=a<t+protocol.horizon]
    for i,a,b in matched:
        history = a-protocol.horizon-protocol.warmup_seconds
        if history>=0 and not any(c<a and d+protocol.postictal_seconds>history for c,d in events):
            return 1,'eligible_preictal',i
    if matched:
        return -1,'preictal_of_ineligible_event',None
    return 0,'observed_horizon_negative',None

"""严格读取、原子写入与取消检查，避免留下可误用的半成品。"""
import json
import os
from pathlib import Path
import tempfile
import hashlib

class Cancelled(RuntimeError):
    pass

def check_cancel(cancel):
    if cancel is not None and cancel.is_set():
        raise Cancelled('任务已取消；本任务的中间文件不会作为完成结果使用。')

def input_file(value, suffix=None):
    if not isinstance(value, (str, Path)) or not str(value).strip():
        raise ValueError('请填写有效的本地文件完整路径。')
    path = Path(value).expanduser().resolve()
    if not path.is_file():
        raise ValueError(f'文件不存在：{path}')
    if suffix and path.suffix.lower() != suffix:
        raise ValueError(f'需要 {suffix} 文件：{path.name}')
    return path

def read_json(path, limit=4*1024*1024):
    path = input_file(path, '.json')
    if path.stat().st_size > limit:
        raise ValueError('JSON 文件超过大小限制。')
    def reject_constant(value):
        raise ValueError(f'JSON 含非法非有限数：{value}')
    def unique(pairs):
        result = {}
        for key, value in pairs:
            if key in result:
                raise ValueError(f'JSON 重复键：{key}')
            result[key] = value
        return result
    return json.loads(path.read_text(encoding='utf-8-sig'),
                      parse_constant=reject_constant, object_pairs_hook=unique)

def atomic_bytes(path, payload):
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    fd, temporary = tempfile.mkstemp(prefix='.partial-', dir=path.parent)
    try:
        with os.fdopen(fd, 'wb') as stream:
            stream.write(payload)
            stream.flush()
            os.fsync(stream.fileno())
        os.replace(temporary, path)
    finally:
        if os.path.exists(temporary):
            os.unlink(temporary)

def write_json(path, value):
    atomic_bytes(path, json.dumps(value, ensure_ascii=False, indent=2,
                                 allow_nan=False).encode('utf-8'))

def file_hash(path):
    digest = hashlib.sha256()
    with Path(path).open('rb') as stream:
        for block in iter(lambda: stream.read(1024*1024), b''):
            digest.update(block)
    return digest.hexdigest()

  function svgNode(tag, attributes, text) {
    const element = document.createElementNS("http://www.w3.org/2000/svg", tag);

  function renderChart() {
    const patient = byId("preview-patient").value;
    const record = byId("preview-record").value;
    const all = state.preview.filter((point) => String(point.patient ?? "未注明") === patient && String(point.record_id ?? point.record ?? "未注明记录") === record)
      .filter((point) => point.time_seconds !== null && point.time_seconds !== "" &&
        point.time_seconds !== undefined && Number.isFinite(Number(point.time_seconds)) && point.probability !== null &&
        point.probability !== "" && Number.isFinite(Number(point.probability)))
      .sort((a, b) => Number(a.time_seconds) - Number(b.time_seconds));
    const container = byId("probability-chart");
    state.chartPoints = all;
    byId("curve-point").max = String(Math.max(0, all.length - 1));
    byId("curve-point").value = "0";
    byId("curve-point").disabled = !all.length;
    if (!all.length) {
      replace(container, [node("p", "", "该患者没有可绘制的有效风险评分时间点。")]);
      byId("chart-caption").textContent = "";
      return;
    }
    const stride = Math.max(1, Math.ceil(all.length / 600));
    const points = all.filter((_, index) => index % stride === 0);
    if (points[points.length - 1] !== all[all.length - 1]) points.push(all[all.length - 1]);
    const width = 880, height = 230, left = 52, right = 21, top = 20, bottom = 41;
    const min = Number(all[0].time_seconds), max = Number(all[all.length - 1].time_seconds);
    const range = Math.max(1, max - min);
    const x = (seconds) => left + (Number(seconds) - min) / range * (width - left - right);
    const y = (probability) => top + (1 - Math.max(0, Math.min(1, Number(probability)))) * (height - top - bottom);
    const svg = svgNode("svg", { viewBox: `0 0 ${width} ${height}`, role: "img", "aria-label": `${patient}风险评分时间曲线；共 ${all.length} 个有效预览时间点` });
    svg.append(svgNode("title", {}, `${patient}窗口风险评分，横轴为记录内时间，纵轴为未校准风险分数`));
    [0, .25, .5, .75, 1].forEach((value) => {
      svg.append(svgNode("line", { x1: left, x2: width - right, y1: y(value), y2: y(value), stroke: "#e5eee8", "stroke-width": 1 }));
      svg.append(svgNode("text", { x: left - 13, y: y(value) + 4, fill: "#96aaa0", "text-anchor": "end", "font-size": 10 }, value.toFixed(2)));
    });
    for (let index = 0; index <= 4; index++) {
      const seconds = min + range * index / 4;
      svg.append(svgNode("text", { x: x(seconds), y: height - 19, fill: "#96aaa0", "text-anchor": "middle", "font-size": 10 }, `${(seconds / 60).toFixed(1)} 分`));
    }
    const threshold = state.previewThreshold;
    svg.append(svgNode("line", { x1: left, x2: width - right, y1: y(threshold), y2: y(threshold), stroke: "#d7b57b", "stroke-width": 1.3, "stroke-dasharray": "5 5" }));
    // Reset the line across record boundaries and gaps instead of implying continuous observation.
    const positiveGaps = all.slice(1).map((point, index) => Number(point.time_seconds) - Number(all[index].time_seconds)).filter((gap) => gap > 0);
    positiveGaps.sort((a, b) => a - b);
    const typicalGap = positiveGaps.length ? positiveGaps[Math.floor(positiveGaps.length / 2)] : 10;
    const gapLimit = Math.max(30, typicalGap * stride * 3);
    let path = "", previous = null, gapCount = 0;
    points.forEach((point) => {
      const record = point.record_id ?? point.record;
      const previousRecord = previous?.record_id ?? previous?.record;
      const changedRecord = previous && record !== undefined && previousRecord !== undefined && record !== previousRecord;
      const discontinuity = previous && (changedRecord || Number(point.time_seconds) - Number(previous.time_seconds) > gapLimit);
      if (discontinuity) gapCount++;
      path += `${!previous || discontinuity ? "M" : "L"}${x(point.time_seconds).toFixed(2)} ${y(point.probability).toFixed(2)} `;
      previous = point;
    });
    svg.append(svgNode("path", { d: path, fill: "none", stroke: "#169d7d", "stroke-width": 2, "stroke-linecap": "round", "stroke-linejoin": "round" }));
    if (points.length === 1) svg.append(svgNode("circle", { cx: x(points[0].time_seconds), cy: y(points[0].probability), r: 3, fill: "#169d7d" }));
    state.chartGeometry = { x, y, top, bottom: height - bottom };
    svg.append(svgNode("line", { id: "curve-cursor", x1: left, x2: left, y1: top, y2: height - bottom, stroke: "#6886a3", "stroke-width": 1, "stroke-dasharray": "3 4" }));
    svg.append(svgNode("circle", { id: "curve-marker", cx: left, cy: y(points[0].probability), r: 4, fill: "#087c6f", stroke: "white", "stroke-width": 2 }));
    replace(container, [svg]);
    renderCurvePoint();
    byId("chart-caption").textContent = `实线为窗口风险评分，虚线为分类阈值 ${threshold.toFixed(2)}。返回 ${all.length.toLocaleString("zh-CN")} 个有效时间点${stride > 1 ? `，绘图抽样 ${points.length} 点` : ""}。${gapCount ? `检测到 ${gapCount} 处时间或记录空档，曲线分段绘制。` : ""}完整预测请下载 CSV。`;
  }

    anchor.href = url;

  function exportPatients() {

  }

  function renderWaveform(preview) {
    const channels = Array.isArray(preview.channels) ? preview.channels : [];
    const samples = Array.isArray(preview.samples) ? preview.samples : [];
    if (!channels.length || samples.length !== channels.length) return null;
    const width = 900, rowH = 62, left = 78, right = 112, top = 24, bottom = 34;
    const height = channels.length * rowH + top + bottom;
    const seconds = Number(preview.seconds) || 10;
    const plotW = width - left - right;
    const t = Array.isArray(preview.t) ? preview.t.map(Number) : [];
    const svg = svgNode("svg", { viewBox: `0 0 ${width} ${height}`, role: "img", "aria-label": `${preview.patient || "患者"} ${preview.record || ""} 输入信号波形，${seconds} 秒，${channels.length} 导联` });
    svg.append(svgNode("title", {}, `${preview.patient || ""} ${preview.record || ""} 输入信号波形，横轴为记录内相对时间（秒），纵轴为微伏`));
    svg.append(svgNode("rect", { x: 0, y: 0, width, height, fill: "#fbfdfc" }));
    channels.forEach((channel, index) => {
      const row = samples[index] || [];
      if (!row.length) return;
      const y0 = top + index * rowH + rowH / 2;
      const values = row.map(Number);
      const min = Math.min(...values), max = Math.max(...values);
      const center = (min + max) / 2, span = Math.max(1e-9, max - min) / 2;
      const scale = rowH * 0.38 / span;
      svg.append(svgNode("line", { x1: left, x2: width - right, y1: y0, y2: y0, stroke: "#e5eee8", "stroke-width": 1 }));
      if (center - span < 0 && center + span > 0) {
        svg.append(svgNode("line", { x1: left, x2: width - right, y1: y0 - (0 - center) * scale, y2: y0 - (0 - center) * scale, stroke: "#d7b57b", "stroke-width": .7, "stroke-dasharray": "3 4" }));
      }
      let path = "";
      const step = Math.max(1, Math.ceil(t.length / 700));
      for (let i = 0; i < t.length; i += step) {
        const x = left + t[i] / seconds * plotW;
        const y = y0 - (values[i] - center) * scale;
        path += `${i === 0 ? "M" : "L"}${x.toFixed(2)} ${y.toFixed(2)} `;
      }
      svg.append(svgNode("path", { d: path, fill: "none", stroke: "#169d7d", "stroke-width": 1.2, "stroke-linejoin": "round", "stroke-linecap": "round" }));
      svg.append(svgNode("text", { x: left - 9, y: y0 + 4, fill: "#385953", "text-anchor": "end", "font-size": 11, "font-weight": 600 }, channel));
      svg.append(svgNode("text", { x: width - right + 7, y: y0 + 4, fill: "#96aaa0", "font-size": 9 }, `${min.toFixed(0)}…${max.toFixed(0)} μV`));
    });
    const xTicks = 5;
    for (let i = 0; i <= xTicks; i++) {
      const at = seconds * i / xTicks;
      const x = left + at / seconds * plotW;
      svg.append(svgNode("text", { x, y: height - 12, fill: "#96aaa0", "text-anchor": "middle", "font-size": 10 }, `${at.toFixed(1)} s`));
    }
    svg.append(svgNode("text", { x: left, y: top - 5, fill: "#8d9f9b", "font-size": 10 }, `文件内 ${Number(preview.start_second ?? 0).toFixed(0)} s 起 · ${preview.fs || 128} Hz 因果滤波 · 微伏（μV）`));
    return svg;
  }

  function renderTask(explicitSelection = false) {
    const job = state.jobs.find((item) => item.id === state.selectedId);
    byId("task-panel").hidden = !job;
    if (!job) return;
    const signature = JSON.stringify(job);
    if (signature === state.selectedSignature) return;
    state.selectedSignature = signature;
    const changedSelection = state.displayedId !== job.id;
    state.displayedId = job.id;
    const status = statusOf(job), active = isActive(job);

    byId("task-status").className = status === "completed" ? "tag tag-green" : "tag";

  function mergeJobs(jobs) {
    if (!Array.isArray(jobs)) throw new Error("任务列表格式不正确，请检查本地服务版本。");
    state.jobs = jobs.filter((job) => job && typeof job.id === "string");
    if (!state.selectedId && state.jobs.length) state.selectedId = state.jobs[0].id;
    if (state.selectedId && !state.jobs.some((job) => job.id === state.selectedId)) state.selectedId = state.jobs[0]?.id || null;
    renderJobs();
    renderTask();
    renderOverview();
    state.jobs.forEach(applyCompletedOutputs);
    updateControls();
  }

  function selectJob(id, scroll = false) {
    state.selectedId = id;
    state.selectedSignature = "";
    renderJobs();
    renderTask(true);
    if (scroll) byId("task-panel").scrollIntoView({ behavior: "smooth", block: "start" });
  }

  function schedulePoll(delay) {
    window.clearTimeout(state.pollTimer);
    if (!state.disposed) state.pollTimer = window.setTimeout(poll, delay);
  }

  async function initializeInfo() {
    const info = await fetchJSON("/api/info");
    if (!info || typeof info.csrf_token !== "string" || !info.csrf_token) throw new Error("本地服务缺少会话校验信息，请重新启动软件。");
    state.info = info;
    window.dispatchEvent(new CustomEvent("eeg:ready", { detail: info }));
    setDefaultPaths(info.default_paths);
    if (typeof info.project_path === "string" && !byId("project-path").dataset.edited) byId("project-path").value = info.project_path;
  }

  async function poll() {
    if (state.polling || state.disposed) return;
    state.polling = true;
    try {
      if (!state.info) await initializeInfo();
      const response = await fetchJSON("/api/jobs");
      mergeJobs(response.jobs);
      state.failures = 0;
      setConnection(true);
    } catch (error) {
      state.failures += 1;
      // A restarted local process has a new CSRF token; refresh metadata on reconnect.
      state.info = null;
      setConnection(false, `${error.message} 任务记录仍保留，连接恢复后会自动更新。`);
    } finally {
      state.polling = false;
      const delay = state.failures ? Math.min(30000, 2500 * 2 ** Math.min(state.failures - 1, 4)) :
        document.hidden ? 12000 : hasActiveJob() ? 1200 : 5000;
      schedulePoll(delay);
    }
  }

  async function startJob(action, params, button) {
    if (state.submitting || hasActiveJob()) {
      toast("已有任务正在运行。请等待完成，或取消后再开始。", true);
      return;
    }
    if (!state.connected || !state.info?.csrf_token) {
      toast("尚未连接本地服务，请刷新或检查软件启动窗口。", true);
      return;
    }
    state.submitting = true;
    updateControls();
    const original = button?.textContent;
    if (button) button.textContent = "正在提交…";
    try {
      const response = await fetchJSON("/api/jobs", { method: "POST", body: JSON.stringify({ action, params }) });
      const job = response.job || response;
      if (!job || typeof job.id !== "string") throw new Error("服务未返回有效任务标识，请刷新任务记录后确认。");
      state.createdIds.add(job.id);
      state.jobs = [job, ...state.jobs.filter((item) => item.id !== job.id)];
      state.selectedId = job.id;
      state.selectedSignature = "";
      renderJobs();
      renderTask();
      if (action !== "connectivity") byId("task-panel").scrollIntoView({ behavior: "smooth", block: "start" });
      toast("任务已提交，运行进度和结果会自动更新。");
      schedulePoll(300);
      return job;
    } catch (error) {
      toast(error.message, true);
      // A network timeout may follow a successful server-side enqueue; refresh before retrying.
      schedulePoll(500);
    } finally {
      state.submitting = false;
      if (button && original) button.textContent = original;
      updateControls();
    }
  }

  function bindForm(id, action, fields) {
    byId(id).addEventListener("submit", (event) => {
      event.preventDefault();
      const form = event.currentTarget;
      if (!form.reportValidity()) return;
      const data = new FormData(form), params = {};
      fields.forEach((field) => { params[field] = String(data.get(field) || "").trim(); });
      if (action === "evaluate") {
        params.threshold = Number(params.threshold);
        if (!Number.isFinite(params.threshold) || params.threshold < 0 || params.threshold > 1) {
          toast("分类阈值必须是 0 到 1 之间的数字。", true);
          return;
        }
      }
      if (fields.some((field) => field !== "threshold" && !params[field])) {
        toast("请填写完整的本机文件路径。", true);
        return;
      }
      startJob(action, params, form.querySelector("button[type=submit]"));
    });
  }

  async function cancelJob() {
    const job = state.jobs.find((item) => item.id === state.selectedId);
    if (!job || !isActive(job) || state.cancelPending) return;
    state.cancelPending = true;
    updateControls();
    try {
      await fetchJSON("/api/cancel", { method: "POST", body: JSON.stringify({ id: job.id }) });
      toast("已提交取消请求。软件会在安全的检查点停止任务。");
      schedulePoll(200);
    } catch (error) {
      toast(error.message, true);
    } finally {
      state.cancelPending = false;
      updateControls();
    }
  }

  });
  byId("result-navigation").querySelectorAll("[data-result]").forEach((button) => button.addEventListener("click", () => {
    byId(button.dataset.result).scrollIntoView({ behavior: "smooth", block: "start" });
  }));
  byId("project-path").addEventListener("input", (event) => { event.currentTarget.dataset.edited = "true"; });
  byId("refresh-button").addEventListener("click", async () => {
    state.failures = 0;
    if (state.polling) { toast("正在刷新，请稍候。 "); return; }
    state.info = null;
    window.clearTimeout(state.pollTimer);
    await poll();
    if (state.connected) toast("已连接本地服务，任务记录已更新。");
  });
  bindForm("prepare-form", "prepare", ["manifest"]);
  bindForm("train-form", "train", ["dataset"]);
  bindForm("predict-form", "predict", ["dataset", "model"]);
  bindForm("evaluate-form", "evaluate", ["predictions", "threshold"]);
  bindForm("history-form", "history", ["project"]);
  window.addEventListener("hashchange", () => showView(window.location.hash.slice(1), false));
  document.addEventListener("visibilitychange", () => { if (!document.hidden) schedulePoll(100); });
  window.addEventListener("pagehide", () => { state.disposed = true; window.clearTimeout(state.pollTimer); });
  window.addEventListener("pageshow", () => { if (state.disposed) { state.disposed = false; schedulePoll(100); } });
  window.EEGApp = Object.freeze({
    submit: (action, params, button) => startJob(action, params, button),
    getInfo: () => state.info,
    getJobs: () => state.jobs.slice(),
    getSelectedJob: () => state.jobs.find(job => job.id === state.selectedId) || null,
    showView,
  });
  showView(window.location.hash.slice(1) || "overview", false);
  updateControls();
  poll();
})();

function status(message, error = false) { $('status').textContent = message; $('status').classList.toggle('error', error); }
function validateData(data) {
  if (!data || data.schema_version !== 1 || !Array.isArray(data.nodes) || data.nodes.length !== 18 || !Array.isArray(data.frames) || !data.frames.length || data.frames.length > 60) throw new Error('网络文件结构不兼容，应包含 18 路导联和 1–60 个时间窗。');
  if (!data.source || !['edf','synthetic_demo'].includes(data.source.kind)) throw new Error('网络数据缺少可识别的数据来源。');
  const ids = new Set();
  data.nodes.forEach(n => {
    if (typeof n.id !== 'string' || typeof n.label !== 'string' || ids.has(n.id) || !Array.isArray(n.position) || n.position.length !== 3 || !n.position.every(v=>finite(v)&&Math.abs(v)<=2) || !Array.isArray(n.electrodes)) throw new Error('导联标识或空间坐标无效。');
    ids.add(n.id);
  });
  ['delta','theta','alpha','beta','gamma'].forEach(id=>{ if (!data.bands?.some(b=>b.id===id && finite(b.low)&&finite(b.high))) throw new Error('频段定义不完整。'); });
  data.frames.forEach((f,index)=>{
    if (!finite(f.start_seconds)||!finite(f.end_seconds)||f.end_seconds<=f.start_seconds || (index && f.start_seconds<data.frames[index-1].end_seconds)) throw new Error('时间窗边界无效或重叠。');
    data.bands.forEach(b=>{
      const matrix=f.matrices?.[b.id];
      if (!Array.isArray(matrix)||matrix.length!==18||matrix.some(row=>!Array.isArray(row)||row.length!==18||row.some(v=>v!==null&&(!finite(v)||v<0||v>1.00001)))) throw new Error('相干性矩阵包含无效数值。');
    });
    const waveform = f.waveform;
    if (!waveform || !Array.isArray(waveform.t) || waveform.t.length>2000 || !waveform.t.length || !waveform.t.every(finite) || !Array.isArray(waveform.samples) || waveform.samples.length!==18 || waveform.samples.some(row=>!Array.isArray(row)||row.length!==waveform.t.length||!row.every(finite))) throw new Error('波形数据长度或数值无效。');
  });
  return data;
}

  const request = ++state.request; loadController?.abort(); loadController = new AbortController(); setPlaying(false);

    state.data=data; state.frame=0; state.selected=-1; state.pair=null; if(jobId)state.lastJob=jobId;
    $('node').innerHTML='<option value="-1">全部导联</option>'+data.nodes.map((n,i)=>`<option value="${i}">${esc(n.label)}</option>`).join('');

    const hash=data.source.sha256 || data.source.file_sha256;

function shares(i,j) { return state.data.nodes[i].electrodes.some(e=>state.data.nodes[j].electrodes.includes(e)); }
function spatial(i,j) {
  const a=state.data.nodes[i].hemisphere,b=state.data.nodes[j].hemisphere;
  return state.hemisphere==='all' || (state.hemisphere==='cross'&&((a==='left'&&b==='right')||(a==='right'&&b==='left'))) || (state.hemisphere==='left'&&a==='left'&&b==='left') || (state.hemisphere==='right'&&a==='right'&&b==='right');
}
function computeEdges() {
  const matrix=frame().matrices[state.band], eligible=[];
  for(let i=0;i<18;i++) for(let j=i+1;j<18;j++) {
    const value=matrix[i][j];
    if(!finite(value)||(!state.shared&&shares(i,j))||!spatial(i,j)) continue;
    eligible.push({i,j,value,shared:shares(i,j)});
  }
  eligible.sort((a,b)=>b.value-a.value||a.i-b.i||a.j-b.j); state.eligible=eligible.length;
  state.edges=eligible.filter(e=>e.value>=state.threshold).slice(0,Math.ceil(eligible.length*state.density/100));
  if (state.focus&&state.selected>=0) state.edges=state.edges.filter(e=>e.i===state.selected||e.j===state.selected);
}

function createCortex() {
  [-1,1].forEach(side=>{
    const geometry=new THREE.SphereGeometry(1,72,48),pos=geometry.attributes.position;
    for(let i=0;i<pos.count;i++){
      let x=pos.getX(i),y=pos.getY(i),z=pos.getZ(i);const theta=Math.atan2(z,x),phi=Math.acos(clamp(y,-1,1));
      const fold=1+.031*Math.sin(theta*14+phi*7)*Math.cos(phi*17)+.018*Math.sin(theta*23-phi*11);
      x=x*.37*fold+side*.35;y=y*.68*fold-.03;z=z*.84*fold;
      pos.setXYZ(i,x,y,z);
    }
    geometry.computeVertexNormals();
    const material=new THREE.MeshPhongMaterial({color:side<0?0x72adc0:0x81afcf,transparent:true,opacity:state.opacity,shininess:48,side:THREE.DoubleSide,depthWrite:false});
    cortexGroup.add(new THREE.Mesh(geometry,material));
    const wire=new THREE.Mesh(geometry.clone(),new THREE.MeshBasicMaterial({color:0x537e93,wireframe:true,transparent:true,opacity:.04,depthWrite:false}));cortexGroup.add(wire);
  });
  // A thin scalp shell communicates sensor placement without claiming anatomy.
  const shell=new THREE.Mesh(new THREE.SphereGeometry(1.02,48,24,0,Math.PI*2,0,Math.PI*.6),new THREE.MeshBasicMaterial({color:0x65b0b8,wireframe:true,transparent:true,opacity:.035,depthWrite:false}));cortexGroup.add(shell);
}
